"""Regression tests for the pre-release hostile review.

Each test pins a behaviour that review demanded and that has shipped
since 0.1.0: fail-closed on unknown ids, consistency warnings on
contradictory answers, the legal-status note on prohibited records, and
Article 26(11) coverage. If any of these regresses, the tool has
reacquired a failure mode a governance tool must not have.
"""

from aiact_triage.engine import classify
from aiact_triage.models import RiskTier, Role, UseCase

=== test_cv_screening_is_high_risk_annex_iii_employment (tests/test_engine.py) ===
def test_cv_screening_is_high_risk_annex_iii_employment():
    case = UseCase(
        name="CV screening tool for graduate recruitment",
        role=Role.DEPLOYER,
        annex_iii_area="employment",
        profiling_of_natural_persons=True,
        workplace_use=True,
    )
    result = classify(case)
    assert result.tier is RiskTier.HIGH_RISK
    # Deployer obligations attach, including informing workers.
    assert "Art. 26(2)" in _articles(result)
    assert "Art. 26(7)" in _articles(result)
    # Annex III dates: original Aug 2026, amended Dec 2027.
    high_risk = [o for o in result.obligations if o.article.startswith("Art. 26")]
    assert all(o.applies_from_original == "2026-08-02" for o in high_risk)
    assert all(o.applies_from_amended == "2027-12-02" for o in high_risk)

=== test_social_scoring_is_prohibited_and_terminal (tests/test_engine.py) ===
def test_social_scoring_is_prohibited_and_terminal():
    case = UseCase(
        name="Citizen trust score",
        role=Role.PROVIDER,
        prohibited_flags=["social_scoring"],
        interacts_with_natural_persons=True,  # must NOT produce Art. 50 duties
    )
    result = classify(case)
    assert result.tier is RiskTier.PROHIBITED
    assert "Art. 50(1)" not in _articles(result)
    prohibition = next(o for o in result.obligations if "5(1)(c)" in o.article)
    assert prohibition.applies_from_original == "2025-02-02"

=== test_provider_chatbot_owes_art_50_1_from_aug_2026 (tests/test_engine.py) ===
def test_provider_chatbot_owes_art_50_1_from_aug_2026():
    case = UseCase(
        name="Chatbot product",
        role=Role.PROVIDER,
        interacts_with_natural_persons=True,
    )
    result = classify(case)
    assert result.tier is RiskTier.TRANSPARENCY
    duty = next(o for o in result.obligations if o.article == "Art. 50(1)")
    assert duty.applies_from_original == "2026-08-02"
    assert duty.applies_from_amended is None  # unchanged by the Omnibus

**Conclusion.** Design effective: the suite collects 38 tests across four files
in a clean environment, covers all three classification gates the control names,
and carries the regression file `tests/test_review_regressions.py`, whose nine
tests pin behaviours a pre-release hostile review demanded, which is the
mechanism by which a found error becomes a permanent case rather than a one-off
fix. Each of the three tests spot-read asserts a classification outcome
directly on the tier the engine returns, `RiskTier.HIGH_RISK`,
`RiskTier.PROHIBITED` and `RiskTier.TRANSPARENCY`, and then asserts the article
and date detail behind it, including the negative assertion that a prohibited
case produces no Article 50 duty. The collected count of 38 is recorded here as
an inventory fact; its agreement with the README claim is CA-08's test, not
this one. Whether the suite passes is the operating test and is not addressed
here.
